// webpack.config.js
// Custom Expo Webpack configuration to alias React Native's asset registry

const { withExpoWeb } = require('@expo/webpack-config');

module.exports = async function (env, argv) {
  const config = await withExpoWeb(env, argv);

  // Alias the missing asset registry module to Expo's implementation
  config.resolve.alias = {
    ...config.resolve.alias,
    '@react-native/assets-registry/registry': require.resolve('expo-asset/src/AssetRegistry'),
  };

  return config;
};
